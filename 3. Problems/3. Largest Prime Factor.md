> [!QUESTION] #3 Largest Prime Factor
>  The prime factors of 13195 are 5, 7, 13 and 29. 
> What is the largest prime factor of the number 600851475143 ?

```python
def _prime(p: int) -> bool:
    # returns True if p is prime, False otherwise
    n = 3
    prime = True
    if p == 2:
        return True
    elif p%2==0:
        prime = False
    while n < p:
        if p%n==0:
            prime = False
        n += 2
    return prime
```



```python
def largest_prime_factor(N: int) -> int:
    # returns the largest prime factor of N
    n = 2
    primes = []
    steps = 0
    while n < N:
        if (N%n == 0) and (_prime(n)):
            # print(f'{n}:{N}->{N/n}')
            N = N/n
            primes.append(n)
        n += 1
        steps += 1
    return primes
```
